// Simple Pong game (() => { const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); const playerScoreEl = document.getElementById('playerScore'); const cpuScoreEl = document.getElementById('cpuScore'); const startBtn = document.getElementById('startBtn'); const pauseBtn = document.getElementById('pauseBtn'); const resetBtn = document.getElementById('resetBtn'); const WIDTH = canvas.width; const HEIGHT = canvas.height; // Game settings const PADDLE_WIDTH = 12; const PADDLE_HEIGHT = 100; const PADDLE_SPEED = 6; const CPU_SPEED = 4.2; const BALL_RADIUS = 8; const INITIAL_BALL_SPEED = 4.0; const MAX_BALL_SPEED = 12; const WIN_SCORE = 10; // Game state let running = false; let paused = false; let animationId = null; const player = { x: 20, y: (HEIGHT - PADDLE_HEIGHT) / 2, width: PADDLE_WIDTH, height: PADDLE_HEIGHT, dy: 0 }; const cpu = { x: WIDTH - 20 - PADDLE_WIDTH, y: (HEIGHT - PADDLE_HEIGHT) / 2, width: PADDLE_WIDTH, height: PADDLE_HEIGHT }; const ball = { x: WIDTH / 2, y: HEIGHT / 2, r: BALL_RADIUS, speed: INITIAL_BALL_SPEED, vx: 0, vy: 0 }; let scores = { player: 0, cpu: 0 }; // Helpers function clamp(v, a, b) { return Math.max(a, Math.min(b, v)); } function resetBall(direction = (Math.random() < 0.5 ? 1 : -1)) { ball.x = WIDTH / 2; ball.y = HEIGHT / 2; ball.speed = INITIAL_BALL_SPEED; const angle = (Math.random() * 0.6 - 0.3); // -0.3..0.3 radians ball.vx = direction * ball.speed * Math.cos(angle); ball.vy = ball.speed * Math.sin(angle); } function drawNet() { ctx.fillStyle = 'rgba(255,255,255,0.06)'; const step = 16; for (let y = 0; y < HEIGHT; y += step * 2) { ctx.fillRect(WIDTH / 2 - 1, y, 2, step); } } function draw() { // clear ctx.clearRect(0, 0, WIDTH, HEIGHT); // net drawNet(); // paddles ctx.fillStyle = '#39ff14'; roundRect(ctx, player.x, player.y, player.width, player.height, 4, true, false); roundRect(ctx, cpu.x, cpu.y, cpu.width, cpu.height, 4, true, false); // ball ctx.beginPath(); ctx.fillStyle = '#e6eef8'; ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2); ctx.fill(); // scores are in DOM, but draw a small center hint ctx.fillStyle = 'rgba(255,255,255,0.06)'; ctx.font = '12px sans-serif'; ctx.textAlign = 'center'; ctx.fillText('Pong — Left: Player • Right: Computer', WIDTH / 2, 18); } // Rounded rect helper function roundRect(ctx, x, y, w, h, r) { if (w < 2 * r) r = w / 2; if (h < 2 * r) r = h / 2; ctx.beginPath(); ctx.moveTo(x + r, y); ctx.arcTo(x + w, y, x + w, y + h, r); ctx.arcTo(x + w, y + h, x, y + h, r); ctx.arcTo(x, y + h, x, y, r); ctx.arcTo(x, y, x + w, y, r); ctx.closePath(); ctx.fill(); } function update() { if (!running || paused) return; // update player position by keys player.y += player.dy; player.y = clamp(player.y, 0, HEIGHT - player.height); // CPU movement - follow ball with limited speed const cpuCenter = cpu.y + cpu.height / 2; const diff = ball.y - cpuCenter; if (Math.abs(diff) > 8) { cpu.y += clamp(diff * 0.04, -CPU_SPEED, CPU_SPEED); } cpu.y = clamp(cpu.y, 0, HEIGHT - cpu.height); // move ball ball.x += ball.vx; ball.y += ball.vy; // top/bottom collision if (ball.y - ball.r <= 0) { ball.y = ball.r; ball.vy = -ball.vy; } else if (ball.y + ball.r >= HEIGHT) { ball.y = HEIGHT - ball.r; ball.vy = -ball.vy; } // paddle collision function function paddleCollision(p) { // AABB check if ( ball.x - ball.r < p.x + p.width && ball.x + ball.r > p.x && ball.y - ball.r < p.y + p.height && ball.y + ball.r > p.y ) { return true; } return false; } // left paddle collision if (ball.vx < 0 && paddleCollision(player)) { // prevent sticking ball.x = player.x + player.width + ball.r; collideWithPaddle(player, 1); } // right paddle collision if (ball.vx > 0 && paddleCollision(cpu)) { ball.x = cpu.x - ball.r; collideWithPaddle(cpu, -1); } // scoring if (ball.x + ball.r < 0) { // CPU scores scores.cpu += 1; cpuScoreEl.textContent = scores.cpu; checkGameOver(); resetBall(1); } else if (ball.x - ball.r > WIDTH) { // Player scores scores.player += 1; playerScoreEl.textContent = scores.player; checkGameOver(); resetBall(-1); } } function checkGameOver() { if (scores.player >= WIN_SCORE || scores.cpu >= WIN_SCORE) { running = false; paused = true; // announce winner via alert (simple) const winner = scores.player >= WIN_SCORE ? 'Player' : 'Computer'; setTimeout(() => { alert(`${winner} wins! Final score: Player ${scores.player} — CPU ${scores.cpu}`); }, 50); } } function collideWithPaddle(paddle, directionSign) { // directionSign: 1 if ball comes from left (bounce right), -1 if from right (bounce left) // compute where the ball hit the paddle relative to its center to vary angle const paddleCenter = paddle.y + paddle.height / 2; const relativeIntersectY = (ball.y - paddleCenter) / (paddle.height / 2); const maxBounceAngle = (75 * Math.PI) / 180; // in radians const bounceAngle = clamp(relativeIntersectY, -1, 1) * maxBounceAngle; // increase speed slightly ball.speed = Math.min(ball.speed + 0.35, MAX_BALL_SPEED); // set velocities based on calculated angle ball.vx = directionSign * ball.speed * Math.cos(bounceAngle); ball.vy = ball.speed * Math.sin(bounceAngle); // add a tiny nudge based on paddle move (player dy) ball.vy += player.dy * 0.15; } // input handling window.addEventListener('keydown', (e) => { if (e.key === 'ArrowUp') { player.dy = -PADDLE_SPEED; e.preventDefault(); } else if (e.key === 'ArrowDown') { player.dy = PADDLE_SPEED; e.preventDefault(); } else if (e.key === ' ' ) { // space toggles pause togglePause(); e.preventDefault(); } }); window.addEventListener('keyup', (e) => { if ((e.key === 'ArrowUp' && player.dy < 0) || (e.key === 'ArrowDown' && player.dy > 0)) { player.dy = 0; } }); // mouse control: move left paddle with mouse within canvas canvas.addEventListener('mousemove', (e) => { const rect = canvas.getBoundingClientRect(); const mouseY = e.clientY - rect.top; player.y = clamp(mouseY - player.height / 2, 0, HEIGHT - player.height); }); // buttons startBtn.addEventListener('click', () => { if (!running) { startGame(); } else if (paused) { paused = false; pauseBtn.textContent = 'Pause'; } }); pauseBtn.addEventListener('click', () => { togglePause(); }); resetBtn.addEventListener('click', () => { resetGame(); }); function togglePause() { if (!running) return; paused = !paused; pauseBtn.textContent = paused ? 'Resume' : 'Pause'; } function startGame() { if (!running) { running = true; paused = false; scores.player = 0; scores.cpu = 0; playerScoreEl.textContent = scores.player; cpuScoreEl.textContent = scores.cpu; player.y = (HEIGHT - player.height) / 2; cpu.y = (HEIGHT - cpu.height) / 2; resetBall(Math.random() < 0.5 ? 1 : -1); pauseBtn.textContent = 'Pause'; loop(); } else { paused = false; } } function resetGame() { running = false; paused = false; scores.player = 0; scores.cpu = 0; playerScoreEl.textContent = 0; cpuScoreEl.textContent = 0; player.y = (HEIGHT - player.height) / 2; cpu.y = (HEIGHT - cpu.height) / 2; resetBall( (Math.random() < 0.5) ? 1 : -1 ); draw(); if (animationId) cancelAnimationFrame(animationId); } function loop() { update(); draw(); animationId = requestAnimationFrame(loop); } // initial draw resetBall(1); draw(); // Allow the game to start by clicking Start // but also allow pressing Space (toggle) after start // Expose simple API on window for debugging window._pong = { startGame, resetGame }; })();
